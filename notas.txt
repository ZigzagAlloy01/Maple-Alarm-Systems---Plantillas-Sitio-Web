(function () {
    function actualizarLogosPago() {
        const amexImg = document.querySelector(
            'img[alt="American Express"]'
        );

        if (!amexImg) return;

        const container = amexImg.closest('.gap-1.flex-wrap.d-flex');

        if (!container) return;

        // Evitar ejecutar la modificación varias veces
        if (container.dataset.paymentLogosUpdated === 'true') {
            return;
        }

        container.dataset.paymentLogosUpdated = 'true';

        // Conservar únicamente American Express
        Array.from(container.children).forEach(span => {
            const img = span.querySelector('img');

            if (!img || img.alt !== 'American Express') {
                span.remove();
            }
        });

        // Crear Visa
        const visaSpan = document.createElement('span');

        visaSpan.className =
            'position-relative d-block rounded overflow-hidden z-1 shadow-sm';

        visaSpan.setAttribute('data-bs-toggle', 'tooltip');
        visaSpan.setAttribute('data-bs-placement', 'top');
        visaSpan.setAttribute('data-bs-delay', '0');
        visaSpan.setAttribute('aria-label', 'Visa');
        visaSpan.setAttribute('data-bs-original-title', 'Visa');

        visaSpan.innerHTML = `
            <img
                src="TU_URL_VISA.svg"
                class="img img-fluid"
                alt="Visa"
                loading="lazy"
                style=""
            >
        `;

        // Crear Mastercard
        const mastercardSpan = document.createElement('span');

        mastercardSpan.className =
            'position-relative d-block rounded overflow-hidden z-1 shadow-sm';

        mastercardSpan.setAttribute('data-bs-toggle', 'tooltip');
        mastercardSpan.setAttribute('data-bs-placement', 'top');
        mastercardSpan.setAttribute('data-bs-delay', '0');
        mastercardSpan.setAttribute('aria-label', 'Mastercard');
        mastercardSpan.setAttribute('data-bs-original-title', 'Mastercard');

        mastercardSpan.innerHTML = `
            <img
                src="TU_URL_MASTERCARD.svg"
                class="img img-fluid"
                alt="Mastercard"
                loading="lazy"
                style=""
            >
        `;

        container.appendChild(visaSpan);
        container.appendChild(mastercardSpan);
    }

    if (document.readyState === 'loading') {
        document.addEventListener(
            'DOMContentLoaded',
            actualizarLogosPago
        );
    } else {
        actualizarLogosPago();
    }
})();
